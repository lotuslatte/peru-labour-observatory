Reproducir esta cifra

Código completo para obtener el valor nacional de 2025 a partir de los microdatos de la ENAHO que publica el INEI. Cada receta carga los módulos, construye las variables, declara el diseño muestral y estima. Las versiones que se muestran se ejecutaron sobre los microdatos y devuelven la cifra publicada.

library(haven); library(dplyr); library(survey)
ruta <- "."  # folder with the ENAHO 2025 .dta files
leer <- function(f) read_dta(file.path(ruta, f)) |> zap_labels() |> rename_with(tolower)
among <- function(x, m) ifelse(m %in% 1, as.integer(coalesce(as.logical(x), FALSE)), NA_integer_)

d <- leer("enaho01a-2025-500.dta") |> filter(p204 %in% 1 | is.na(p204), p208a >= 14, fac500a > 0)

d <- d |> mutate(
  ocupado = as.integer(ocu500 %in% 1),
  # CIIU rev. 4 division (numeric codes drop the leading zero: 111 is class 0111)
  division = floor(p506r4 / 100),
  sector9 = case_when(ocupado != 1 ~ NA_real_, division %in% 1:3 ~ 1, division %in% 5:9 ~ 2, division %in% 10:33 ~ 3, division %in% 35:39 ~ 4, division %in% 41:43 ~ 5, division %in% 45:47 ~ 6, division %in% c(49:53, 58:63) ~ 7, division %in% c(55:56, 90:98) ~ 8, division %in% c(64:66, 68:75, 77:82, 84:88, 99) ~ 9)
)

options(survey.lonely.psu = "adjust")  # singleton strata centred on the grand mean
d <- d |> mutate(estrato_dept = paste(substr(ubigeo, 1, 2), estrato))
des <- svydesign(ids = ~conglome, strata = ~estrato_dept, weights = ~fac500a, data = d, nest = TRUE)

r <- svymean(~factor(sector9), subset(des, ocupado %in% 1 & !is.na(sector9)), na.rm = TRUE)
est <- 100 * unname(coef(r))
print(round(100 * coef(r), 1))

# Rama de actividad, ENAHO 2025. Observatorio Laboral del Perú.
import numpy as np
import pandas as pd

RUTA = "."  # folder with the ENAHO 2025 .dta files
CLAVES = ['conglome', 'vivienda', 'hogar', 'codperso']


def leer(archivo, columnas=None):
    t = pd.read_stata(f"{RUTA}/{archivo}", convert_categoricals=False, columns=columnas)
    t.columns = t.columns.str.lower()
    return t


def among(x, m):
    """x as 0/1 where m equals 1, missing elsewhere."""
    return x.astype(float).where(m.eq(1))


def se_total(z):
    """Taylor standard error of a total of scores z: PSUs within strata, singleton strata centred."""
    p = pd.DataFrame({"h": d.estrato_dept, "c": d.conglome, "z": z}).groupby(["h", "c"]).z.sum().reset_index()
    n = p.groupby("h").z.transform("size")
    dev = p.z - p.groupby("h").z.transform("mean")
    v = (n / (n - 1) * dev ** 2)[n > 1].sum() + ((p.z[n == 1] - p.z.mean()) ** 2).sum()
    return v ** 0.5


def ratio(y, m):
    """Weighted mean of y where m is true, with its design standard error."""
    m = m & y.notna()
    w = d.fac500a.where(m, 0)
    r = (w * y.fillna(0)).sum() / w.sum()
    return r, se_total(w * (y.fillna(0) - r) / w.sum())


d = leer("enaho01a-2025-500.dta")
d = d[d.p204.fillna(1).eq(1) & (d.p208a >= 14) & (d.fac500a > 0)].reset_index(drop=True)
d["estrato_dept"] = d.ubigeo.str[:2] + "-" + d.estrato.astype(int).astype(str)

d["ocupado"] = d.ocu500.eq(1).astype(int)
# CIIU rev. 4 division (numeric codes drop the leading zero: 111 is class 0111)
division = np.floor(d.p506r4 / 100)
grupos = {1: [(1, 3)], 2: [(5, 9)], 3: [(10, 33)], 4: [(35, 39)], 5: [(41, 43)], 6: [(45, 47)], 7: [(49, 53), (58, 63)], 8: [(55, 56), (90, 98)], 9: [(64, 66), (68, 75), (77, 82), (84, 88), (99, 99)]}
d["sector9"] = np.nan
for g, rangos in grupos.items():
    for lo, hi in rangos: d.loc[division.between(lo, hi), "sector9"] = g
d["sector9"] = d.sector9.where(d.ocupado.eq(1))

categorias = {1: 'Agropecuario y pesca', 2: 'Minería', 3: 'Manufactura', 4: 'Electricidad y agua', 5: 'Construcción', 6: 'Comercio', 7: 'Transporte y comunicaciones', 8: 'Alojamiento, restaurantes y otros servicios', 9: 'Servicios profesionales, públicos y sociales'}
est = {}
for k, nombre in categorias.items():
    e, se = ratio(d.sector9.eq(k).astype(float).where(d.sector9.notna()), d.ocupado.eq(1))
    est[k] = 100 * e
    print(f"{nombre}: {100 * e:.1f}%   error estándar: {100 * se:.2f}")
